912.排序数组

题目：912. 排序数组open in new window

难度：中等

归并排序

时间复杂度：O(nlogn)

空间复杂度：O(n)

class Solution:
    def merge(self, left, right):
        i = 0
        j = 0
        merged = []
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                merged.append(left[i])
                i += 1
            else:
                merged.append(right[j])
                j += 1
        merged += left[i: ]
        merged += right[j: ]
        return merged

    def merge_sort(self, nums):
        if len(nums) < 2:
            return nums

        mid = len(nums) // 2
        left = self.merge_sort(nums[: mid])
        right = self.merge_sort(nums[mid: ])
        return self.merge(left, right)

    def sortArray(self, nums: List[int]) -> List[int]:
        return self.merge_sort(nums)

二路快排

时间复杂度：O(nlogn)

空间复杂度：O(1)

import random
class Solution:
    def swap(self, nums, a, b):
        nums[a], nums[b] = nums[b], nums[a]

    def partition(self, nums, low, high):
        rand_index = random.randint(low, high)
        self.swap(nums, low, rand_index)
        pivot = nums[low]
        i = low + 1
        j = high

        while i <= j:
            while i <= high and nums[i] < pivot:
                i += 1
            while j >= 0 and nums[j] > pivot:
                j -= 1
            if i >= j:
                break
            self.swap(nums, i, j)
            i += 1
            j -= 1
        self.swap(nums, j, low)
        return j

    def quick_sort(self, nums, low, high):
        if low < high:
            p = self.partition(nums, low, high)
            self.quick_sort(nums, low, p - 1)
            self.quick_sort(nums, p + 1, high)

    def sortArray(self, nums: List[int]) -> List[int]:
        self.quick_sort(nums, 0, len(nums)-1)
        return nums

堆排序

时间复杂度：O(nlogn)

空间复杂度：O(1)

class Solution:
    def heapify(self, nums, i, end):
        largest = i
        left = i * 2 + 1
        right = i * 2 + 2

        if left < end and nums[largest] < nums[left]:
            largest = left
        if right < end and nums[largest] < nums[right]:
            largest = right

        # 若最大的值不是nums[i]
        if largest != i:
            nums[i], nums[largest] = nums[largest], nums[i]
            self.heapify(nums, largest, end)

    def sortArray(self, nums: List[int]) -> List[int]:
        if not nums:
            return nums

        n = len(nums)
        # 构建大根堆
        for i in range(n // 2 - 1, -1, -1):
            self.heapify(nums, i, n)

        # 排序：每次从堆顶取出最大元素，放在末尾（与末尾交换）
        # 然后调整堆顶，且注意数组的范围缩减了
        for i in range(n - 1, 0, -1):
            nums[0], nums[i] = nums[i], nums[0]
            self.heapify(nums, 0, i)
        return nums

class Solution:

        nums[a], nums[b] = nums[b], nums[a]

    def partition(self, nums, low, high):

        self.swap(nums, low, high)

        i = low + 1
        j = high

                i += 1
            while j >= 0 and nums[j] > pivot:
                j -= 1
            if i >= j:
                break
            self.swap(nums, i, j)
            i += 1
            j -= 1
        self.swap(nums, low, j)
        return j

    def sort(self, nums, k, low, high):
        while True:
            p = self.partition(nums, low, high)
            if p == k:
                return nums[k]
            elif p > k:
                high = p - 1
            elif p < k:
                low = p + 1

    def findKthLargest(self, nums: List[int], k: int) -> int:
        return self.sort(nums, len(nums) - k, 
                            0, len(nums) - 1)

class Solution:
    def swap(self, nums, a, b):
        nums[a], nums[b] = nums[b], nums[a]

    def heapify(self, nums, i, n):
        largest = i
        left = i * 2 + 1
        right = i * 2 + 2

        if left < n and nums[left] > nums[largest]:
            largest = left
        if right < n and nums[right] > nums[largest]:
            largest = right

        if largest != i:
            self.swap(nums, i, largest)
            self.heapify(nums, largest, n)

    def findKthLargest(self, nums: List[int], k: int) -> int:
        n = len(nums)
        for i in range(n // 2 - 1, -1, -1):
            self.heapify(nums, i, n)

        for i in range(n-1, n-1-k, -1):
            self.swap(nums, i, 0)
            self.heapify(nums, 0, i)

        return nums[-k]

class Solution:
    def removeDuplicates(self, nums: List[int]) -> int:
        # 两个指针，一个从0开始，一个从1开始
        i = 0
        j = 1
        while j < len(nums):
            # 若j指针的数==i指针的数，j移动
            if nums[j] == nums[i]:
                j += 1
            else:  # 否则i先移动，然后赋值过来
                i += 1
                nums[i] = nums[j]
        return i + 1  # 注意i的含义

    j = 0
    k = len(nums) - 1
    while j <= k:
        if nums[j] == 0:
            nums[i], nums[j] = nums[j], nums[i]

            j += 1  # 注意这里有j指针的更新

            j += 1
        else:

难度：中等

class Solution:
    def searchMatrix(self, matrix: List[List[int]], target: int) -> bool:
        low = 0
        high = len(matrix) - 1
        r = low
        while low <= high:
            mid = (low + high) >> 1
            if matrix[mid][0] <= target:
                r = mid
                low = mid + 1
            else:
                high = mid - 1

        low = 0
        high = len(matrix[0]) - 1
        # c = low
        while low <= high:
            mid = (low + high) >> 1
            if matrix[r][mid] < target:
                low = mid + 1
            elif matrix[r][mid] > target:
                high = mid - 1
            else:
                return True
        return False

34. 在排序数组中查找元素的第一个和最后一个位置

题目：34. 在排序数组中查找元素的第一个和最后一个位置open in new window

难度：中等

弄懂本题需要对二分查找懂透，对于一个序列：[0, 0, 0, 1, 1]，需要找到0和1的分界处，按照三段式的二分查找模板（nums[mid]与target的三种大小关系分开讨论）：

若希望找到分界处的0，即索引为2的位置，临界情况为[0, 1]，左指针此时指向0，右指针指向1，此时mid指向0，最终退出循环时一定成为了：左指针指向1，右指针指向0，我们希望得到0，则返回right；

若希望找到分界处的1，即索引为3的位置，临界情况为[0, 1]，左指针此时指向0，右指针指向1，此时mid指向0，最终退出循环时一定成为了：左指针指向1，右指针指向0，我们希望得到0，则返回left；

除了上面两个返回结果不同外，还有一个区别，即当nums[mid]==target时，若希望找到0的位置，则此时应该左移右指针；若希望找到1的位置，则此时应该右移左指针。

class Solution:
    def search_left(self, nums, low, high, target):
        while low <= high:
            mid = (low + high) >> 1
            if nums[mid] == target:
                high = mid - 1  # 区别1
            elif nums[mid] > target:
                high = mid - 1
            else:
                low = mid + 1
        return low  # 区别2

    def search_right(self, nums, low, high, target):
        while low <= high:
            mid = (low + high) >> 1
            if nums[mid] == target:
                low = mid + 1  # 区别1
            elif nums[mid] > target:
                high = mid - 1
            else:
                low = mid + 1
        return high  # 区别2

    def searchRange(self, nums: List[int], target: int) -> List[int]:
        if len(nums) == 0:
            return [-1, -1]
        left = self.search_left(nums, 0, len(nums) - 1, target)
        right = self.search_right(nums, 0, len(nums) - 1, target)
        if left > right:
            return [-1, -1]
        return [left, right]

33. 搜索旋转排序数组

题目：33. 搜索旋转排序数组 - 力扣（LeetCode）open in new window

难度：中等

数组被划分为了两个递增子数组，左边序列的第一位大于右边序列的最后一位：

中间的情况比较复杂，中间涉及到两个数组的切换，在实现的时候不需要考虑中间，只需要将两边考虑清楚即可，即mid落在左子序列还是右子序列中，分别落在两边时，该如何变化指针。

class Solution:
    def search(self, nums: List[int], target: int) -> int:
        if not nums:
            return -1
        left = 0
        right = len(nums) - 1
        while left <= right:
            mid = (left + right) >> 1
            if nums[mid] == target:
                return mid
            elif nums[mid] >= nums[0]:  # mid处于左边递增序列
                if nums[0] <= target < nums[mid]:
                    right = mid - 1
                else:
                    left = mid + 1
            else: # mid处于右边的递增序列
                if nums[mid] < target <= nums[len(nums)-1]:
                    left = mid + 1
                else:
                    right = mid - 1
        return -1

153.寻找旋转排序数组中的最小值

题目：153. 寻找旋转排序数组中的最小值 - 力扣（LeetCode）open in new window

难度：中等

本题的数组大小顺序和上一题差不多，只是多了一种情况：可能是递增的，这个可以利用首元素和末元素大小判断。

class Solution:
    def findMin(self, nums: List[int]) -> int:
        # 整个数组单调递增：
        if nums[0] < nums[len(nums)-1] or len(nums) == 1:
            return nums[0]

        # 局部单调递增：
        left = 0
        right = len(nums) - 1
        while left <= right:
            mid = (left + right) >> 1
            if nums[mid] >= nums[0]:
                left = mid + 1
            elif nums[mid] < nums[0]:
                right = mid - 1
        # 假设输入的是[3,4,5,1,2]
        # 临界情况下，left和right分别指向[5,1]
        # 再继续模拟，left会指向1，right指向5然后退出循环
        # 我们希望返回left位置的值
        return nums[left]

394. 字符串解码

题目：394. 字符串解码 - 力扣（LeetCode）open in new window

难度：中等

使用栈来操作。

时间复杂度：记解码后得出的字符串长度为 S，除了遍历一次原字符串 s，我们还需要将解码后的字符串中的每个字符都入栈，并最终拼接进答案中，故渐进时间复杂度为 O(S+∣s∣)

class Solution:
    def decode_num(self, nums):
        n = 0
        for i in range(len(nums)):
            n += 10 ** (len(nums) - i - 1) * nums[i]
        return n

    def decodeString(self, s: str) -> str:
        stack = deque()
        i = 0
        while i < len(s):
            # 若遇到数字
            # 将后续的数字全部加入nums，并解析成数并入栈
            if ord('0') <= ord(s[i]) <= ord('9'):
                nums = []
                while ord('0') <= ord(s[i]) <= ord('9'):
                    nums.append(int(s[i]))
                    i += 1
                # 解析数字
                n = self.decode_num(nums)
                stack.append(n)

            # 若遇到字母或'['
            # 入栈
            elif ord('a') <= ord(s[i]) <= ord('z') or s[i] == '[':
                stack.append(s[i])
                i += 1

            # 若遇到']'
            # 先让s指针往后移动
            # 再不断弹出栈内元素并加入列表，直到遇到'['
            # 再弹出栈顶元素，此时一定是一个数
            # 需要将列表字符翻转一下*该数，入栈
            else:  
                i += 1  # 对s的遍历往后走一位
                sup = []
                while stack[-1] != '[':
                    c = stack.pop()
                    sup.append(c)
                stack.pop()  # 将'['弹出
                n = stack.pop()
                stack.append(''.join(sup[::-1] * n))
        # 最后将栈内所有字符组成字符串
        ans = ''.join(stack)
        return ans

739.每日温度

题目：739. 每日温度 - 力扣（LeetCode）open in new window

难度：中等

单调栈

时间复杂度：O(n)

空间复杂度：O(n)

class Solution:
    def dailyTemperatures(self, temperatures: List[int]) -> List[int]:
        stack = deque([[temperatures[0], 0]])

        ans = [0] * len(temperatures)
        for i in range(len(temperatures)):
            t = temperatures[i]
            if not stack:  # 栈为空，入栈
                stack.append([t, i])
            else:  # 栈不为空，入栈之前，将所有小于t的弹出
                while stack and stack[-1][0] < t:
                    temp, ind = stack.pop()
                    ans[ind] = i - ind
                stack.append([t, i])
        return ans

55.跳跃游戏

class Solution:
    def canJump(self, nums: List[int]) -> bool:
        max_pos = nums[0]
        for i in range(len(nums)):
            # 当前位置能跳到最远的地方
            max_pos = max(max_pos, nums[i] + i)
            # 若该最远地方可以跳出最后一个
            if max_pos >= len(nums) - 1:
                return True
            # 若现在已经到达能挑到的最远位置，但还未到最后一个
            if i == max_pos and i < len(nums) - 1:
                return False
        return True

300.最长递增子序列

题目：300. 最长递增子序列open in new window

难度：中等

方法1：动归

时间复杂度：O(n^2)

空间复杂度：O(n^2)

思路是，一开始每一个数都可以作为子序列的末位，然后遍历nums，当遍历的数nums[i]作为前面子序列nums[: i]的末位时，总长度能否变得更长，然后找出最长的值存在dp[i]内。最后答案不是存在dp[-1]内，而要遍历dp找

class Solution:
    def lengthOfLIS(self, nums: List[int]) -> int:
        # 初始时刻，每一个数都可以作为一个子序列的末位
        # 故初始时刻长度都为1
        dp = [1] * len(nums)

        # 每一个数作为末位时，前面的数对应的子序列何时可以增加长度
        for i in range(len(nums)):
            for j in range(i):
                if nums[j] < nums[i]:
                    # dp[i]可能已经加了前面某一个dp[j]
                    # 所以dp[i]要将本身和dp[j]+1比较
                    dp[i] = max(dp[i], dp[j]+1)
        return max(dp)  # 注意不是dp[-1]

方法2：二分查找+贪心

class Solution:
    def lengthOfLIS(self, nums: List[int]) -> int:
        # 思路：用d数组存最长的递增子序列
        # 如何使得其最长呢？那就是每次增加的时候
        # 选择数值最接近的，慢慢增加，这样才能最长（贪心）
        length = 0
        d = [0] * (len(nums) + 1)
        d[0] = float('-inf')  # 保证第一个数可以加入
        for i in range(len(nums)):
            # 若nums[i]大于最长递增子序列的末位，直接加入
            if nums[i] > d[length]:
                length += 1
                d[length] = nums[i]

            # 若遇到nums[i]小于最长递增子序列的末位
            # 用nums[i]去覆盖掉d里最后一个不小于nums[i]的数
            # 因为d一定是递增的，故可以使用二分法查找
            else: 
                low = 1  # 0是最小值位
                high = length
                while low <= high:
                    mid = (low + high) >> 1
                    if d[mid] > nums[i]:
                        high = mid - 1
                    elif d[mid] < nums[i]:
                        low = mid + 1
                    else:
                        low = mid
                        break
                d[low] = nums[i]
        return length

64.最小路径和

题目：64. 最小路径和 - 力扣（LeetCode）open in new window

难度：中等

class Solution:
    def minPathSum(self, grid: List[List[int]]) -> int:
        m = len(grid)
        n = len(grid[0])
        dp = [[0] * n for _ in range(m)]

        # init
        dp[0][0] = grid[0][0]
        for i in range(1, m):
            dp[i][0] = grid[i][0] + dp[i-1][0]
        for i in range(1, n):
            dp[0][i] = grid[0][i] + dp[0][i-1]

        for i in range(1, m):
            for j in range(1, n):
                dp[i][j] = min(dp[i-1][j], dp[i][j-1]) + grid[i][j]
        return dp[m-1][n-1]

class Solution:
    def longestCommonSubsequence(self, text1: str, text2: str) -> int:
        n1 = len(text1)
        n2 = len(text2)
        dp = [[0] * (n2 + 1) for _ in range(n1 + 1)]

        # init

        # update
        for i in range(1, n1 + 1):
            for j in range(1, n2 + 1):
                if text1[i-1] == text2[j-1]:
                    dp[i][j] = dp[i-1][j-1] + 1
                else:
                    dp[i][j] = max(dp[i-1][j], dp[i][j-1])
        return dp[n1][n2]

198.打家劫舍

难度：中等

class Solution:
    def rob(self, nums: List[int]) -> int:
        dp = [[0, 0] for i in range(len(nums))]
        dp[0][1] = nums[0]  # 抢劫, dp[0][0]是不抢劫
        for i in range(1, len(nums)):
            # 今天没抢，可以昨天也没抢，或者昨天抢了
            dp[i][0] = max(dp[i-1][0], dp[i-1][1])
            # 今天抢了，可以昨天没抢
            dp[i][1] = dp[i-1][0] + nums[i]
        return max(dp[-1][0], dp[-1][1])

279.完全平方数

题目：279. 完全平方数 - 力扣（LeetCode）open in new window

难度：中等

class Solution:
    def numSquares(self, n: int) -> int:
        # 思路：使用dp[i]表示 和为i的完全平方数的最小数量
        # 则最后返回dp[n]即可
        dp = [0] * (n + 1)
        # dp[1] = 1

        # 更新dp时，dp[i] = dp[i-j**2]+1
        # 其中j<=i，+1表示加了一个j**2
        # 因为要使得dp[i-j**2]尽可能小，所以要遍历[1,i]
        for i in range(1, n+1):
            min_val = float('inf')
            for j in range(1, i):
                j_square = j ** 2
                if j_square > i:
                    break
                min_val = min(min_val, dp[i-j_square])
            if min_val == float('inf'):  
                min_val = 0
            dp[i] = min_val + 1
        return dp[n]

994. 腐烂的橘子

题目：994. 腐烂的橘子 - 力扣（LeetCode）open in new window

难度：中等

class Solution:
    def check(self, grid, i, j):
        if i < 0 or i >= self.m or j < 0 or j >= self.n:
            return False
        if grid[i][j] == 2 or grid[i][j] == 0:
            return False
        return True

    def orangesRotting(self, grid: List[List[int]]) -> int:
        self.m = len(grid)
        self.n = len(grid[0])
        dir_x = [-1, 1, 0, 0]
        dir_y = [0, 0, -1, 1]
        self.ans = 0
        q = deque()
        self.good = 0
        # 因为题目要扩散的次数，则只能用广度优先遍历
        # 先遍历一遍，将好橘子数量记好，每一次扩散需要减少该值，最终判断该值是否为0，来判断是否所有橘子都腐烂
        # 同时要将坏橘子加入队列，模拟初始状态
        for i in range(self.m):
            for j in range(self.n):
                if grid[i][j] == 1:
                    self.good += 1
                elif grid[i][j] == 2:
                    q.append([i, j])
        # 每一次将坏橘子弹出，上下左右的坏橘子加入队列，同时标记该橘子坏了
        # 即good -= 1, q.append(new_x, new_y), grid[new_x][new_y]=2
        while q:
            sz = len(q)
            for _ in range(sz):
                x, y = q.popleft()
                # 将上下左右的新鲜橘子坐标加入self.q
                # 并将grid[i][j] = 2，self.good -= 1
                for i in range(len(dir_x)):
                    new_x = x+dir_x[i]
                    new_y = y+dir_y[i]
                    if self.check(grid, new_x, new_y):
                        grid[new_x][new_y] = 2
                        self.good -= 1  # 好橘子烂了
                        q.append([new_x, new_y])
            if q:  # 最后这个地方有点坑，因为要下一次还可以扩散才加时间
                self.ans += 1
        return self.ans if self.good == 0 else -1

207.课程表

题目：207. 课程表 - 力扣（LeetCode）open in new window

难度：中等

class Solution:
    def build(self, numCourses, prerequisites):
        graph = {}
        # 每一门课程作为一个节点，课程的序号从0~n-1
        for i in range(numCourses):
            graph[i] = []
        # 将每一个节点的后继节点加入
        for r in prerequisites:
            graph[r[1]].append(r[0])
        return graph

    def check(self, graph, numCourses, prerequisites):
        """判断是否有环"""
        # 统计每一个节点的入度
        for r in prerequisites:
            self.inDegree[r[0]] += 1
        # 将入度为0的节点作为起点
        q = deque()
        for i in range(numCourses):
            if self.inDegree[i] == 0:
                q.append(i)
        # 开始分解图：将入度为0的去除，并减少其后继节点的入度
        cnt = 0
        while q:
            sz = len(q)
            cnt += sz
            for _ in range(sz):
                node = q.popleft()
                for nex in graph[node]:
                    self.inDegree[nex] -= 1
                    if self.inDegree[nex] == 0:
                        q.append(nex)
        # 若最终减少的节点个数少于总节点个数-1(最后有一个节点剩余)，说明存在环
        if cnt < numCourses:
            self.hasCycle = True

    def canFinish(self, numCourses: int, prerequisites: List[List[int]]) -> bool:
        self.inDegree = [0] * numCourses
        self.hasCycle = False
        graph = self.build(numCourses, prerequisites)
        self.check(graph, numCourses, prerequisites)
        return not self.hasCycle  # 是否能完成学习

class Solution:
    def inorderTraversal(self, root: Optional[TreeNode]) -> List[int]:
        stack = deque()
        ans = []
        while stack or root:
            while root:
                stack.append(root)
                root = root.left
            root = stack.pop()
            ans.append(root.val)
            root = root.right
        return ans

146.LRU缓存

题目：146. LRU 缓存open in new window

难度：中等

双向链表+哈希表

class Node():
    def __init__(self, key=-1, val=-1, next=None, prev=None):
        self.key = key
        self.val = val
        self.next = next
        self.prev = prev


class LRUCache:

    def __init__(self, capacity: int):
        self.capacity = capacity
        self.size = 0
        self.head = Node()
        self.tail = Node()
        self.head.next = self.tail
        self.tail.prev = self.head
        self.mp = {}  # key: node

    def get(self, key: int) -> int:
        if key in self.mp:
            self.remove(key)
            node = self.mp[key]
            self.add2head(node)
            return node.val
        else:
            return -1

    def put(self, key: int, value: int) -> None:
        if key not in self.mp:
            node = Node(key, value)
            self.mp[key] = node
            self.add2head(node)
            self.size += 1
            if self.size > self.capacity:
                node = self.removeTail()
                self.size -= 1
                self.mp.pop(node.key)
        else:
            node = self.mp[key]
            node.val = value
            self.remove(key)
            self.add2head(node)

    def remove(self, key):
        node = self.mp[key]
        node.prev.next = node.next
        node.next.prev = node.prev

    def removeTail(self):
        node = self.tail.prev
        self.remove(node.key)
        return node

    def add2head(self, node):
        node.next = self.head.next
        self.head.next.prev = node
        self.head.next = node
        node.prev = self.head

class MinStack:

    def __init__(self):
        self.min = float('inf')
        self.stack = []
        self.min_stack = []

    def push(self, val: int) -> None:
        self.stack.append(val)
        if not self.min_stack:
            self.min_stack.append(val)
        else:
            self.min_stack.append(min(val, self.min_stack[-1]))

    def pop(self) -> None:
        self.stack.pop()
        self.min_stack.pop()

    def top(self) -> int:
        val = self.stack[-1]
        return val

    def getMin(self) -> int:
        return self.min_stack[-1]

560.和为k的子数组

题目：560. 和为 K 的子数组 - 力扣（LeetCode）open in new window

难度：中等

使用前缀和来降低时间复杂度，并使用mp来存储所有 前缀和: 数量 对，因为任何一个子数组和 sum(nums[i: j]) 可以表示为：sum(nums[: j]) - k：

class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        ans = 0
        prefix = 0  # sum(nums[: i])
        mp = {0: 1}  # val: n, 值为val的 前缀和 的个数为n

        for i in range(len(nums)):
            prefix += nums[i]
            sub_arr_sum = prefix - k
            if sub_arr_sum in mp:
                ans += mp[sub_arr_sum]
            # 注意mp里的key都是前缀和
            mp[prefix] = mp.get(prefix,0) + 1
        return ans
