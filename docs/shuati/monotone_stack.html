单调栈

INFO

何时使用单调栈？通常是一维数组，要寻找任一个元素的右边或者左边第一个比自己大或者小的元素的位置，此时我们就要想到可以用单调栈了。时间复杂度为O(n)。代码随想录 (programmercarl.com)open in new window

739. 每日温度

题目：739. 每日温度 - 力扣（LeetCode）open in new window

难度：中等

方法1：双指针暴力

超时。

class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        int p = 0, q = 0;
        vector<int> ans(temperatures.size(), 0);
        while (p < temperatures.size()) {
            q = p + 1;
            while (q < temperatures.size()) {
                if (temperatures[q] > temperatures[p]) {
                    ans[p] = q - p;
                    break;
                }
                q ++;
            }
            p++;
        }
        return ans;
    }
};

方法2：单调栈

class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        stack<int> st;
        st.push(0);
        vector<int> ans(temperatures.size(), 0);
        for (int i = 1; i < temperatures.size(); i++) {
            if (temperatures[st.top()] >= temperatures[i]) {
                st.push(i);
            }
            else {
                while (!st.empty() && temperatures[st.top()] < temperatures[i]) {
                    ans[st.top()] = i - st.top();
                    st.pop();
                }
                st.push(i);  // 没进入循环说明此时的T[i]比栈顶温度大或栈为空，需要入栈
            }
        }
        return ans;
    }
};

更简洁的写法：

class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        stack<int> st;
        st.push(0);
        vector<int> ans(temperatures.size(), 0);
        for (int i = 1; i < temperatures.size(); i++) {
            while (!st.empty() && temperatures[st.top()] < temperatures[i]) {
                ans[st.top()] = i - st.top();
                st.pop();
            }
            st.push(i);  // 没进入循环说明此时的T[i]比栈顶温度大或栈为空，需要入栈
        }
        return ans;
    }
};

时间复杂度：O(n)，空间复杂度：O(n)。

方法3：暴力2

这是官方提供的暴力解法，但是能通过。思路是：给定i，求在[temperatures[i]+1, 100]之间的温度（>temperatures[i]）在temperatures列表中的索引，并保留最小索引值。

那么就要求外层循环遍历temperatures，内层循环遍历[temperatures[i]+1, 100]，在内层循环中要同时记录下在temperatures内且大于temperatures[i]且索引大于i的索引值，这里有点绕，其实就是用一个哈希表来存温度-索引对。

那就有两种做法，第一种是先遍历一遍temperatures，构造该哈希表，然后再在后续的二重循环的内层循环内用一个if判断语句筛选一下，但是尝试之后发现这不对，因为temperatures里可能有重复值，会将哈希表里的值不断覆盖。

第二种做法，不提前构造map，而是一边遍历temperatures一边更新map，而为了找到未来时刻比temperatures[i]更大的温度值，这就要外层循环反向遍历，这样可以实现的功能是每次访问temperatures[i]时，其后续的温度-索引哈希已经构建完毕：

class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        vector<int> ans(temperatures.size(), 0);
        vector<int> map(101, INT_MAX);
        for (int i = temperatures.size() - 1; i >= 0; i--) {
            int max_idx = INT_MAX;
            for (int t = temperatures[i] + 1; t < 101; t++) {
                max_idx = min(max_idx, map[t]);
            }
            if (max_idx != INT_MAX) ans[i] = max_idx - i;
            map[temperatures[i]] = i;
        }
        return ans;
    }
};

496. 下一个更大元素1

题目：496. 下一个更大元素 Iopen in new window

难度：简单

方法1：暴力

先找到nums1[i]在nums2中对应的元素的位置，然后再在nums2中找后一个比nums1[i]大的值。

class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        vector<int> ans(nums1.size(), -1);
        for (int i = 0; i < nums1.size(); i++) {
            int index = 0;
            for (int j = 0; j < nums2.size(); j++) {
                if (nums2[j] == nums1[i]) {
                    index = j; 
                    break;
                }
            }
            for (int j = index+1; j < nums2.size(); j++) {
                if (nums2[j] > nums2[index]) {
                    ans[i] = nums2[j];
                    break;
                }
            }
        }
        return ans;
    }
};

时间复杂度：O(nm)。

方法2：哈希表

第一种方法的不足之处在于，每次要找到nums1[i]在nums2中对应的位置这一步需要遍历一下nums2，所以这个方法先提前将nums2中的value-index的哈希表创建好：

class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        vector<int> ans(nums1.size(), -1);
        unordered_map<int, int>map;

        for (int i = 0; i < nums2.size(); i++) {
            map[nums2[i]] = i;
        }
        for (int i = 0; i < nums1.size(); i++) {
            for (int j = map[nums1[i]] + 1; j < nums2.size(); j++) {
                if (nums2[j] > nums1[i]) {
                    ans[i] = nums2[j];
                    break;
                }
            }
        }
        return ans;
    }
};

方法3：单调栈+哈希表

如果要使用单调栈，我的一个想法是，先用一个二重循环构造nums1中的元素到nums2的hashmap，然后再使用二重循环：外层循环遍历nums1，找到nums1[i]在nums2的位置，内层循环....这不是又回到了方法2。。。

单调栈为何叫单调栈，是因为栈内元素始终保持单调，打破单调的元素早就被pop出了。这要求最好是记录的全局的数列表，而不是从数列表的某个地方开始，这处理的是局部的数列表。

所以官方给的思路是：先使用单调栈求出nums2中的每个数的符合要求的结果，同时构造哈希表，然后再遍历nums1，将哈希表的结果取出构造ans。

class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        vector<int> ans(nums1.size(), -1);
        stack<int> st;
        unordered_map<int, int> map;

        for (int i = 0; i < nums2.size(); i++) {
            if (st.empty()) {
                st.push(i);
                map[nums2[i]] = -1;
            }
            else if (nums2[st.top()] >= nums2[i]) {
                map[nums2[i]] = -1;
                st.push(i);
            }
            else {
                while (!st.empty() && nums2[st.top()] < nums2[i]) {
                    map[nums2[st.top()]] = nums2[i];
                    st.pop();
                }
                map[nums2[i]] = -1;
                st.push(i);
            }
        }
        for (int i = 0; i < nums1.size(); i++) ans[i] = map[nums1[i]];
        return ans;
    }
};

其中的map[nums2[i]] = -1;非常容易忘，还可以将代码简化一下：

class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        vector<int> ans(nums1.size(), -1);
        stack<int> st;
        unordered_map<int, int> map;

        for (int i = 0; i < nums2.size(); i++) {
            while (!st.empty() && nums2[st.top()] < nums2[i]) {
                map[nums2[st.top()]] = nums2[i];
                st.pop();
            }
            map[nums2[i]] = -1;
            st.push(i);
        }
        for (int i = 0; i < nums1.size(); i++) ans[i] = map[nums1[i]];
        return ans;
    }
};

代码随想录给了另一个思路：先构建nums1[i]-i的哈希表，然后使用单调栈对nums2内元素求解的同时，利用哈希表更新ans：

class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        vector<int> ans(nums1.size(), -1);
        stack<int> st;
        unordered_map<int, int> map;

        for (int i = 0; i < nums1.size(); i++) {
            map[nums1[i]] = i;
        }

        for (int i = 0; i < nums2.size(); i++) {
            int num = nums2[i];
            while (!st.empty() && num > st.top()) {
                if (map.count(st.top()) > 0) {  // in nums1
                    ans[map[st.top()]] = num;
                }
                st.pop();
            }
            st.push(num);
        }
        return ans;
    }
};

503.下一个更大元素2

题目：503. 下一个更大元素 IIopen in new window

难度：中等

方法1：暴力解法

循环每一个元素，找到符合的结果存入ans。时间复杂度：O(n^2).

class Solution {
public:
    vector<int> nextGreaterElements(vector<int>& nums) {
        vector<int> ans(nums.size(), -1);
        for (int i = 0; i < nums.size(); i++) {
            int begin = (i+1) % nums.size();
            while (begin != i) {
                if (nums[begin] > nums[i]) {
                    ans[i] = nums[begin];
                    break;
                }
                begin = (begin + 1) % nums.size();
            }
        }
        return ans;
    }
};

方法2：单调栈+循环数组

这里先记录一个错误的思路，先用单调栈处理所有的元素，然后结束的时候栈内还有元素，这些元素都是没有找到符合要求的值，那么我们只需要找到栈内这些元素的相应的符合要求的值即可。假设栈内还有n个元素，栈内的值是保持单调递减的，根据题目要求，容易判断出，栈内每一个值的符合要求的值就是栈的第一个数值，所以得到：

class Solution {
public:
    vector<int> nextGreaterElements(vector<int>& nums) {
        vector<int> ans(nums.size(), -1);
        stack<int> st;

        for (int i = 0; i < nums.size(); i++) {
            int num = nums[i];
            while (!st.empty() && num > nums[st.top()]) {
                ans[st.top()] = num;
                st.pop();
            }
            st.push(i);
        }

        int st_size = st.size();
        vector<int> count(st_size);
        for (int i = 0; i < st_size; i++) {
            count[i] = st.top();
            st.pop();
        }
        int max_val = nums[count[count.size()-1]];
        for (int i = 0; i < count.size()-1; i++) {
            if (max_val > nums[count[i]]) ans[count[i]] = max_val;
        }
        return ans;
    }
};

这里错就错在黑体部分的那句话，比如下面的例子：

[1,2,3,2,1]

结束首次遍历nums后，单调栈内的元素为[3, 2, 1]，按照上述的思路，2的符合要求的值为3,这是正确的，但是1符合要求的值却是3，这是错误的，正确的应该是原nums中的第一个2，但是这个2不包含在单调栈内，所以无论如何都无法得到正确结果，这个思路是错的。

正确的解法：

思路为循环数组，循环数组一般不是真正将其展平，而是通过取模来完成。对于每一个数，需要将整个数组都找一遍，因为其满足要求的值可能在nums中排在其左边，按照一般的单调栈的代码结构，外层循环的次数为 nums.size()，但是如果遍历到了最后一个元素，循环下一次就终止了，无法继续找下去，而这最后一个元素最坏的情况还需要再找nums.size()-1次才可以找到，所以外层循环应该为：nums.size() x2-1次，外层循环长度超过了nums.size()，所以访问nums[i]时需要取模：nums[i % nums.size()]：

class Solution {
public:
    vector<int> nextGreaterElements(vector<int>& nums) {
        int n = nums.size();
        vector<int> ans(n, -1);
        stack<int> st;

        for (int i = 0; i < n * 2-1; i++) {
            int num = nums[i % n];
            while (!st.empty() && num > nums[st.top()]) {
                ans[st.top()] = num;
                st.pop();
            }
            st.push(i % n);
        }
        return ans;
    }
};

时间复杂度：O(n)

方法3：单调栈+拼接数组

思路是将两个nums拼接为一个，然后执行单调栈找最大最近值。

class Solution {
public:
    vector<int> nextGreaterElements(vector<int>& nums) {
        int n = nums.size();
        vector<int> nums1(nums.begin(), nums.end());
        nums.insert(nums.end(), nums1.begin(), nums1.end());
        stack<int> st;
        vector<int> ans(nums.size(), -1);  // 使用新数组的长度初始化ans

        for (int i = 0; i < nums.size(); i++) {
            int num = nums[i];
            while (!st.empty() && num > nums[st.top()]) {
                ans[st.top()] = num;
                st.pop();
            }
            cout << endl;
            st.push(i);
        }
        ans.resize(n);
        return ans;
    }
};

这种写法确实比较直观，但做了很多无用操作，例如修改了nums数组，而且最后还要把result数组resize回去。

resize倒是不费时间，是O(1)的操作，但扩充nums数组相当于多了一个O(n)的操作。

其实也可以不扩充nums，而是在遍历的过程中模拟走了两边nums（方法2的做法）。
