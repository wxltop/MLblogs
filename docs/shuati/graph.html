797. 所有可能的路径

题目：797. 所有可能的路径open in new window

难度：中等

方法1：dfs

使用dfs递归来实现图的遍历的时候有这几点需要注意：

  1. 最好定义一个全局的path和results，不然函数传来传去很麻烦；

  2. 递归函数的参数和返回结果应该是什么？

  3. 边界条件（即结束递归的条件）是什么？

  4. 回溯过程的代码结构

这几点确定了很容易写出代码：

class Solution {
public:
    vector<vector<int>> results;
    vector<int> path;

    void dfs(vector<vector<int>>& graph, int node) {
        if (node == graph.size() - 1) {
            results.push_back(path);
            return;
        }
        for (int i = 0; i < graph[node].size(); i++) {
            path.push_back(graph[node][i]);
            dfs(graph, graph[node][i]);
            path.pop_back();
        }
    }
    vector<vector<int>> allPathsSourceTarget(vector<vector<int>>& graph) {
        path.push_back(0);
        dfs(graph, 0);
        return results;
    }
};

时间复杂度：n为图中节点的数量，最坏情况每一个点都可以去往所有其他的点，这里就有一棵递归树，数的最大深度为n，路径数为O(2^n)。每条路径的长度为O(n)，总的时间复杂度为O(n2^n)。

空间复杂度：O(n)

方法2：bfs

bfs是否必须要使用队列来实现呢？不是。如果对于遍历的顺序没有要求，使用栈也可实现，只不过栈可能先顺时针将当前节点的周围节点存起来，然后逆时针再逐一遍历，后来又顺时针存；而队列先进先出，始终保持遍历的顺序的一致性。

本题使用bfs实现的时候需要注意，队列里存放的应该是路径，而不是节点，因为本题需要返回路径，如果不存储，想通过每一次迭代来构造路径是很困难的，无法知道每一条路径的当前节点的之前节点是谁。

class Solution {
public:
    vector<vector<int>> allPathsSourceTarget(vector<vector<int>>& graph) {
        vector<vector<int>> ans;
        queue<vector<int>> q;  // 存放每一条路径
        q.push(vector<int>(1, 0));  // 开始只有一条路径，该路径只有一个点即0
        while (!q.empty()) {
            int npath = q.size();
            for (int i = 0; i < npath; i++) {
                vector<int> path = q.front();  // 取出来一条路径
                q.pop();
                int last_node = path[path.size()-1];
                if (last_node == graph.size()-1) {
                    ans.push_back(path);  // 若路径最后一个节点是图的最后一个节点
                    continue;
                }
                // 如果路径最后一个节点不是图的最后节点
                for (int next: graph[last_node]) {
                    vector<int> new_path = path;
                    new_path.push_back(next);
                    q.push(new_path);
                }
            }
        }
        return ans;
    }
};

时间复杂度应该和dfs一样，空间复杂度需要把每一条路径都存起来，所以也到了O(n2^n)

200. 岛屿数量

题目：200. 岛屿数量open in new window

难度：中等

方法1：dfs

思路是每一次访问一个岛屿之后，将其沉没（设为'0'）。

class Solution {
public:
    void dfs(vector<vector<char>>& grid, int i, int j) {
        // 越界
        if (i < 0 || j < 0 || i >= grid.size() || j >= grid[0].size()) return;
        if (grid[i][j] == '0') return;
        // 如果grid[i][j] == '1'，则可以 return true，只是需要将岛屿淹没后再返回
        grid[i][j] = '0';
        dfs(grid, i-1, j);  // 上
        dfs(grid, i+1, j);  // 下
        dfs(grid, i, j-1);  // 左
        dfs(grid, i, j+1);  // 右
    }
    int numIslands(vector<vector<char>>& grid) {
        int ans = 0;
        for (int i = 0; i < grid.size(); i++) {
            for (int j = 0; j < grid[0].size(); j++) {
                if (grid[i][j] == '1') {
                    ans ++; 
                    dfs(grid, i, j);
                }
            }
        }
        return ans;
    }
};

时间复杂度：O(MN)，其中M 和 N 分别为行数和列数。

空间复杂度：O(MN)，在最坏情况下，整个网格均为陆地，深度优先搜索的深度达到 MN。

上述代码把grid值给改了，还有一个思路是，不修改grid，使用一个新的数组visited来记录访问信息。

class Solution {
public:
    void dfs(vector<vector<char>>& grid, vector<vector<bool>>& visited, int i, int j) {
        // 越界
        if (i < 0 || j < 0 || i >= grid.size() || j >= grid[0].size()) return;
        if (grid[i][j] == '0' || visited[i][j]) {  // 这里 visited[i][j]判断一定要加上，否则会超出递归栈最大深度
            visited[i][j] = true;
            return;
        }
        // 如果grid[i][j] == '1'，则可以 return true，只是需要将岛屿淹没后再返回
        visited[i][j] = true;
        dfs(grid, visited, i-1, j);  // 上
        dfs(grid, visited, i+1, j);  // 下
        dfs(grid, visited, i, j-1);  // 左
        dfs(grid, visited, i, j+1);  // 右
    }
    int numIslands(vector<vector<char>>& grid) {
        int ans = 0;
        vector<vector<bool>> visited(grid.size(), vector<bool>(grid[0].size(), false));
        for (int i = 0; i < grid.size(); i++) {
            for (int j = 0; j < grid[0].size(); j++) {
                if ((!visited[i][j]) && grid[i][j] == '1') {
                    ans ++; 
                    dfs(grid, visited, i, j);
                }
            }
        }
        return ans;
    }
};

方法2：bfs

思路如下：

  1. 逐一遍历每一个点，若该点未被访问且值为'1'，则入队。此时说明发现了新岛屿，ans++；

  2. 要将发现的新岛屿的地域全部找出来，遍历队列里的每一个点，对于每一个点，若其上下左右方向没有超出grid范围且未被访问且值为'1'，则入队；

  3. 当队列为空时，表明一个岛屿已经全部访问完成，继续下一个点。

class Solution {
public:
    int numIslands(vector<vector<char>>& grid) {
        int ans = 0;
        vector<vector<bool>> visited(grid.size(), vector<bool>(grid[0].size(), false));
        queue<pair<int, int>> q;
        for (int i = 0; i < grid.size(); i++) {
            for (int j = 0; j < grid[0].size(); j++) {
                if ((!visited[i][j]) && grid[i][j] == '1') {
                    q.push({i, j});
                    ans ++;
                    while (!q.empty()) {
                        pair<int, int> coor = q.front();
                        int a = coor.first;
                        int b = coor.second;
                        q.pop();
                        if (a > 0 && !visited[a-1][b] && grid[a-1][b] == '1') {
                            q.push({a-1, b});  // 上
                            visited[a-1][b] = true;
                        }
                        if (a < grid.size()-1 && !visited[a+1][b] && grid[a+1][b] == '1') {
                            q.push({a+1, b}); // 下
                            visited[a+1][b] = true;
                        }
                        if (b > 0 && !visited[a][b-1] && grid[a][b-1] == '1') {
                            q.push({a, b-1});  // 左
                            visited[a][b-1] = true;
                        }
                        if (b < grid[0].size()-1 && !visited[a][b+1] && grid[a][b+1] == '1') {
                            q.push({a, b+1}); // 右
                            visited[a][b+1] = true;
                        }
                    }
                }
            }
        }
        return ans;
    }
};

695. 岛屿的最大面积

题目：695. 岛屿的最大面积open in new window

难度：中等

方法1：dfs

会做岛屿数量，这道题应该没啥问题。

class Solution {
public:
    int ans = 0;
    int dfs(vector<vector<int>>& grid, vector<vector<bool>>& visited, int i, int j) {
        if (i < 0 || j < 0 || i >= grid.size() || j >= grid[0].size()) return 0;
        if (grid[i][j] == 0 || visited[i][j] == true) return 0;
        visited[i][j] = true;
        int s = 1;
        s += dfs(grid, visited, i-1, j);
        s += dfs(grid, visited, i+1, j);
        s += dfs(grid, visited, i, j-1);
        s += dfs(grid, visited, i, j+1);
        return s;
    }
    int maxAreaOfIsland(vector<vector<int>>& grid) {
        int m = grid.size(); 
        int n = grid[0].size();
        vector<vector<bool>> visited(m, vector<bool>(n, false));
        int s = 0;

        for (int i = 0; i < m; i ++) {
            for (int j = 0; j < n; j++ ) {
                if (!visited[i][j] && grid[i][j] == 1) {
                    s = dfs(grid, visited, i, j);
                    ans = max(s, ans);
                }
            }
        }
        return ans;
    }
};

时间复杂度：O(MN)，其中M 和 N 分别为行数和列数。我们访问每个网格最多一次。

空间复杂度：O(MN)，在最坏情况下，递归的深度最大可能是整个网格的大小。

方法2：dfs栈实现

使用栈实现dfs，实现起来框架和bfs很像。

class Solution {
public:
    int maxAreaOfIsland(vector<vector<int>>& grid) {
        int m = grid.size(); 
        int n = grid[0].size();
        stack<pair<int, int>> st;
        int ans = 0;
        int di[4] = {-1, 1, 0, 0};
        int dj[4] = {0, 0, -1, 1};

        for (int i = 0; i < m; i ++) {
            for (int j = 0; j < n; j++ ) {
                int s = 0;
                st.push({i, j});
                while (!st.empty()) {
                    pair<int, int> coor = st.top();
                    st.pop();
                    int a = coor.first, b = coor.second;
                    if (a < 0 || b < 0 || a >= m || b >= n || grid[a][b] == 0)
                        continue;
                    s += 1;
                    grid[a][b] = 0;  // 访问过后将其淹没
                    for (int k = 0; k < 4; k++) {
                        st.push({a + di[k], b + dj[k]});
                    }
                }
                ans = max(ans, s);
            }
        }
        return ans;
    }
};

class Solution {
public:
    int maxAreaOfIsland(vector<vector<int>>& grid) {
        int m = grid.size(); 
        int n = grid[0].size();
        vector<vector<bool>> visited(m, vector<bool>(n, false));
        queue<pair<int, int>> q;
        int ans = 0;

        for (int i = 0; i < m; i ++) {
            for (int j = 0; j < n; j++ ) {
                int s = 0;
                if (!visited[i][j] && grid[i][j] == 1) {
                    q.push({i, j});
                    while(!q.empty()) {
                        pair<int, int> coor = q.front();
                        q.pop();

                        int a = coor.first, b = coor.second;
                        visited[a][b] = true;
                        s += 1;

                        if (a > 0 && !visited[a-1][b] && grid[a-1][b] == 1) {
                            q.push({a-1, b});  // 上
                            visited[a-1][b] = true;
                        }
                        if (a < m-1 && !visited[a+1][b] && grid[a+1][b] == 1) {
                            q.push({a+1, b});  // 下
                            visited[a+1][b] = true;
                        }
                        if (b > 0 && !visited[a][b-1] && grid[a][b-1] == 1) {
                            q.push({a, b-1});  // 左
                            visited[a][b-1] = true;
                        }
                        if (b < n-1 && !visited[a][b+1] && grid[a][b+1] == 1) {
                            q.push({a, b+1});  // 右
                            visited[a][b+1] = true;
                        }
                    }
                }
                ans = max(ans, s);
            }
        }
        return ans;
    }
};

时间复杂度：O(MN)，其中M 和 N 分别为行数和列数。

空间复杂度：O(MN)，在最坏情况下，整个网格均为陆地（队列存储），深度优先搜索的深度达到 MN。

1020. 飞地的数量

题目：1020. 飞地的数量open in new window

难度：中等

方法1：dfs

整体框架其实和之前的几道题都一样，可以每次都计量岛屿中陆地的数量，然后判断该岛屿是否临界，如果临界则陆地数量不计入最终的返回结果中。

class Solution {
public:
    int di[4] = {-1, 1, 0, 0};
    int dj[4] = {0, 0, 1, -1};
    bool border(int m, int n, int i, int j) {
        if (i == 0 || j == 0 || i == m-1 || j == n-1) return true;
        return false;
    }
    pair<bool, int> dfs(vector<vector<int>>& grid, vector<vector<bool>>& visited, int i, int j) {
        if (i < 0 || j < 0 || i >= grid.size() || j >= grid[0].size()) return pair<bool, int>({false, 0});
        if (grid[i][j] == 0 || visited[i][j]) return pair<bool, int>({false, 0});

        bool is_border = border(grid.size(), grid[0].size(), i, j);
        visited[i][j] = true;
        pair<bool, int> sub_res({is_border, 1});
        pair<bool, int> tmp;
        for (int k = 0; k < 4; k++) {
            tmp = dfs(grid, visited, i+di[k], j+dj[k]);
            sub_res.first = sub_res.first || tmp.first;
            sub_res.second += tmp.second;
        }
        return sub_res;
    }
    int numEnclaves(vector<vector<int>>& grid) {
        int res = 0;
        int m = grid.size(); 
        int n = grid[0].size();
        vector<vector<bool>> visited(m, vector<bool>(n, false));
        pair<bool, int> tmp({false, 0});

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                tmp = dfs(grid, visited, i, j);
                if (!tmp.first) res += tmp.second;
            }
        }
        return res;
    }
};

时间复杂度：O(mn)，其中 m 和 n 分别是网格 grid 的行数和列数。深度优先搜索最多访问每个单元格一次，需要 O(mn) 的时间，遍历网格统计飞地的数量也需要 O(mn) 的时间。

空间复杂度：O(mn)，其中 m 和 n 分别是网格 grid 的行数和列数。空间复杂度主要取决于 visited 数组和递归调用栈空间，空间复杂度是 O(mn)。

方法2：dfs2

这个思路是：可以先遍历边界，将边界上的岛屿先访问然后再遍历内部的岛屿。虽然时间复杂度和方法1一样，但是这个方法实际上更高效。

class Solution {
public:
    int di[4] = {-1, 1, 0, 0};
    int dj[4] = {0, 0, 1, -1};
    int dfs(vector<vector<int>>& grid, vector<vector<bool>>& visited, int i, int j) {
        if (i < 0 || j < 0 || i >= grid.size() || j >= grid[0].size()) return 0;
        if (grid[i][j] == 0 || visited[i][j]) return 0;

        visited[i][j] = true;
        int cnt = 1;
        for (int k = 0; k < 4; k++) {
            cnt += dfs(grid, visited, i+di[k], j+dj[k]);
        }
        return cnt;
    }
    int numEnclaves(vector<vector<int>>& grid) {
        int res = 0;
        int m = grid.size(); 
        int n = grid[0].size();
        vector<vector<bool>> visited(m, vector<bool>(n, false));
        pair<bool, int> tmp({false, 0});
        for (int i = 0; i < m; i++) {
            dfs(grid, visited, i, 0);  // 0列
            dfs(grid, visited, i, n-1);  // n-1列
        }
        for (int i = 0; i < n; i++) {
            dfs(grid, visited, 0, i);  // 0行
            dfs(grid, visited, m-1, i);  // m-1行
        }

        for (int i = 1; i < m-1; i++) {
            for (int j = 1; j < n-1; j++) {
                res += dfs(grid, visited, i, j);
            }
        }
        return res;
    }
};

方法3：bfs

思路是先将边界的陆地入队，然后逐一遍历队列的元素并更新队列，将边界的岛屿都访问之后，再遍历中间的陆地。

class Solution {
public:
    int di[4] = {-1, 1, 0, 0};
    int dj[4] = {0, 0, 1, -1};
    int numEnclaves(vector<vector<int>>& grid) {
        int res = 0;
        int m = grid.size(); 
        int n = grid[0].size();
        vector<vector<bool>> visited(m, vector<bool>(n, false));
        queue<pair<int, int>> q;

        for (int i = 0; i < m; i++) {
            if (grid[i][0] == 1) q.push({i, 0});
            if (grid[i][n-1] == 1) q.push({i, n-1});
        }
        for (int i = 0; i < n; i++) {
            if (grid[0][i] == 1) q.push({0, i});
            if (grid[m-1][i] == 1) q.push({m-1, i});
        }
        while (!q.empty()) {
            int a = q.front().first, b = q.front().second;
            q.pop();
            visited[a][b] = true;

            for (int i = 0; i < 4; i++) {
                int new_a = a + di[i];
                int new_b = b + dj[i];
                if (new_a < 0 || new_b < 0 || new_a >= m || new_b >= n) continue;
                if (!visited[new_a][new_b] && grid[new_a][new_b] == 1) {
                    q.push({new_a, new_b});
                    visited[new_a][new_b] = true;
                }
            }
        }
        for (int i = 1; i < m-1; i++) {
            for (int j = 1; j < n-1; j++) {
                if (!visited[i][j] && grid[i][j] == 1) res += 1;
            }
        }
        return res;
    }
};

130. 被围绕的区域

题目：130. 被围绕的区域open in new window

难度：中等

方法1：bfs

思路几乎和上一题的思路一样，先遍历边界的陆地，将临界的岛屿访问一次，然后再访问中间部分的陆地，将其改为'X'即可。

class Solution {
public:
    int di[4] = {1, -1, 0, 0};
    int dj[4] = {0, 0, 1, -1};
    void solve(vector<vector<char>>& board) {
        int m = board.size();
        int n = board[0].size();
        vector<vector<bool>> visited(m, vector<bool>(n, false));
        queue<pair<int, int>> q;
        for (int i = 0; i < m; i++) {
            if (board[i][0] == 'O') q.push({i, 0});
            if (board[i][n-1] == 'O') q.push({i, n-1});
        }
        for (int i = 0; i < n; i++) {
            if (board[0][i] == 'O') q.push({0, i});
            if (board[m-1][i] == 'O') q.push({m-1, i});
        }
        while (!q.empty()) {
            int a = q.front().first, b = q.front().second;
            q.pop();
            visited[a][b] = true;

            for (int k = 0; k < 4; k++) {
                int new_a = a + di[k];
                int new_b = b + dj[k];
                if (new_a < 0|| new_b < 0 || new_a >= m || new_b >= n) continue;
                if (!visited[new_a][new_b] && board[new_a][new_b] == 'O') 
                    q.push({new_a, new_b});
            }
        }
        for (int i = 1; i < m-1; i++) {
            for (int j = 1; j < n-1; j++) {
                if (!visited[i][j] && board[i][j] == 'O') board[i][j] = 'X';
            }
        }
    }
};

时间复杂度：O(mn)，其中 m 和 n 分别是网格 grid 的行数和列数。深度优先搜索最多访问每个单元格一次，需要 O(mn) 的时间，遍历网格统计飞地的数量也需要 O(mn) 的时间。

空间复杂度：O(mn)，其中 m 和 n 分别是网格 grid 的行数和列数。空间复杂度主要取决于 visited 数组和递归调用栈空间，空间复杂度是 O(mn)。

417. 太平洋大西洋水流问题

题目：417. 太平洋大西洋水流问题open in new window

难度：中等

方法1：从内到外dfs

思路是若从某个点出发，可以到达大西洋，则返回值1，若可以到达太平洋，则返回值2，若两个都可以到达，则返回值>=3。从每一个点作为起点出发时，都定义一个visited数组，防止重复访问。但是下方代码却超时了：

class Solution {
public:
    int di[4] = {-1, 1, 0, 0};
    int dj[4] = {0, 0, -1, 1};
    int dfs(vector<vector<int>>& heights, vector<vector<bool>>& visited, int pre_h, int i, int j) {
        int m = heights.size();
        int n = heights[0].size();
        if (i < 0 || j < 0) return 1;
        if (i >= m || j >= n) return 2;
        if (visited[i][j]) return 0;
        if (heights[i][j] > pre_h) return 0;

        visited[i][j] = true;
        int res = 0;
        for (int k = 0; k < 4; k++) {
            int tmp = dfs(heights, visited, heights[i][j], i+di[k], j+dj[k]);
            if (tmp != res) res += tmp;
        }
        return res;
    }
    vector<vector<int>> pacificAtlantic(vector<vector<int>>& heights) {
        vector<vector<int>> ans;
        int m = heights.size();
        int n = heights[0].size();
        int pre_h = INT_MAX;
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                vector<vector<bool>> visited(m, vector<bool>(n, false));
                if (dfs(heights, visited, pre_h, i, j) >= 3) ans.push_back({i, j});
            }
        }
        return ans;
    }
};

遍历每一个节点，是 m * n，遍历每一个节点的时候，都要做深搜，深搜的时间复杂度是： m * n。那么整体时间复杂度 就是 O(m2∗n2)，这是一个四次方的时间复杂度。

方法2：从外到内dfs

官方给了一种思路（其实回过头来想居然和前几题的思路一样。。。），从边界的点遍历，每次往同等高度或者更高的地方走，找水流的源头。先从太平洋边界出发，使用一个数组记录能到达的源头，然后从大西洋边界出发，使用一个数组记录能到达的源头，最终遍历每一个点，看谁能同时到达两片海洋。

class Solution {
public:
    int di[4] = {-1, 1, 0, 0};
    int dj[4] = {0, 0, -1, 1};
    void dfs(vector<vector<bool>>& ocean, vector<vector<int>>& heights, int i, int j) {
        int m = ocean.size();
        int n = ocean[0].size();

        if (ocean[i][j]) return;
        ocean[i][j] = true;
        for (int k = 0; k < 4; k++) {
            int new_i = i+di[k];
            int new_j = j+dj[k];
            if (new_i >= 0 && new_i < m && new_j >= 0 && new_j < n) {
                if (heights[new_i][new_j] >= heights[i][j]) 
                    dfs(ocean, heights, new_i, new_j);
            }  
        }
    }
    vector<vector<int>> pacificAtlantic(vector<vector<int>>& heights) {
        vector<vector<int>> ans;
        int m = heights.size();
        int n = heights[0].size();
        vector<vector<bool>> pacific(m, vector<bool>(n, false));
        vector<vector<bool>> atlantic(m, vector<bool>(n, false));

        for (int i = 0; i < m ; i++) {
            dfs(pacific, heights, i, 0);
            dfs(atlantic, heights, i, n-1);
        }
        for (int i = 0; i < n; i++) {
            dfs(pacific, heights, 0, i);
            dfs(atlantic, heights, m-1, i);
        }
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (pacific[i][j] && atlantic[i][j]) {
                    vector<int> cell;
                    cell.emplace_back(i);
                    cell.emplace_back(j);
                    ans.emplace_back(cell);
                }
            }
        }
        return ans;
    }
};

由于有二维数组记录每一个节点是否被遍历过，所以dfs函数时间复杂度为O(mn)，即每一个节点只会走一次，因为有两个海洋数组，所以时间复杂度为O(2*mn)，最后一个二维循环，所以一共O(3mn)。

方法1：bfs

其思路是先构建好图，然后根据广度优先来遍历。参考：

如何构图？

单词作为节点，如果两个单词之间的区别仅仅在一个字母，则两个单词可以连边。由于任意单词的长度都是相同，所以比较起来可以统一。看题目的要求中，如果单词和单词之间来比较从而确定是否要连边，复杂度为O(wordList×word)，单词列表的长度可能很大，很可能超时。所以另一个思路是，逐一修改单词的每一位（遍历25个英文字母），这样时间复杂度为O(26×word)。

在遍历的时候需要注意，要防止进入死循环，即图中有环路出现，需要使用visited数组来记录已经遍历过的节点。

题目细节：

题目很多细节提示很重要：

  • 1 <= wordList.length <= 5000：暗示如果逐一比较可能超时
  • wordList[i].length == beginWord.length：暗示比较单词的时候可以统一逐位比较
  • beginWord、endWord 和 wordList[i] 由小写英文字母组成：暗示可以逐一修改单词的每一位（遍历26个英文字母）来优化时间复杂度
class Solution {
public:
    int ladderLength(string beginWord, string endWord, vector<string>& wordList) {
        unordered_set<string> set(wordList.begin(), wordList.end());
        if (set.find(endWord) == set.end()) return 0;
        unordered_map<string, bool> visited;
        queue<string> q;
        int path = 1;

        q.push(beginWord);
        while (!q.empty()) {
            int len = q.size();
            bool add_path = false;  // 每一层之后path只能一次+1
            for (int i = 0; i < len; i++) {  // 当前层的每一个单词
                string cur_word = q.front();
                q.pop();
                for (int j = 0; j < cur_word.size(); j++) {  // 单词的每一位
                    string new_word = cur_word;
                    for (int c = 0; c < 26; c++) {  // 每一位修改为其他字符
                        new_word[j] = 'a' + c;
                        if (new_word == cur_word) continue;  // 和当前单词相同
                        if (new_word == endWord) return ++path;
                        if (set.find(new_word) != set.end() && !visited[new_word]) {
                            visited[new_word] = true;
                            q.push(new_word);
                            add_path = true;
                        }
                    }
                }
            }
            if (add_path) path ++;
        }
        return 0;
    }
};

时间复杂度：每一个单词最多遍历一遍，每一个单词会衍生出来wordLen x 26个单词，这些单词查询（参与比较），所以时间复杂度为O(wordLen×26)。

空间复杂度：主要和单词列表长度还有单词长度有关，O(wordList×wordLen)

方法2：双向bfs

已知目标顶点的情况下，可以分别从起点和目标顶点（终点）执行广度优先遍历，直到遍历的部分有交集，这是双向广度优先遍历的思想。

使用双向bfs需要注意的地方：

  • 已知目标顶点的情况下，可以分别从起点和目标顶点（终点）执行广度优先遍历，直到遍历的部分有交集。这种方式搜索的单词数量会更小一些；
  • 更合理的做法是，每次从单词数量小的集合开始扩散；
  • 这里 beginVisited 和 endVisited 交替使用，等价于单向 BFS 里使用队列，每次扩散都要加到总的 visited 里。

看代码会发现，总是在交替beginVisited和endVisited，因为我们将beginVisited作为需要接下来扩散的一个方向，而每次选择集合元素更少的可以时间效率更高。

何时退出循环？ 如果beginVisited和endVisited某一个方向扩展不下去了，意味着找不到下一步的路了，退出循环。

何时得到结果？ 如果新构造的单词出现在了endVisited中，则表示路打通了，可以得到结果。而不能和单向bfs一样等于beginWord或者endWord。

class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:
        wordList = set(wordList)
        if endWord not in wordList:
            return 0

        beginVisited = set()  # 记录正向已经遍历过的节点
        beginVisited.add(beginWord)
        endVisited = set()  # 记录反向已经遍历过的节点
        endVisited.add(endWord)
        visited = set()  # 防止重复遍历
        step = 1

        while beginVisited and endVisited:  # 如果某一个方向扩展不了，退出循环
            if (len(beginVisited) > len(endVisited)):  # 每一次挑后继节点少的来继续扩展
                beginVisited, endVisited = endVisited, beginVisited

            next_level = set()
            add_step = False
            for word in beginVisited:  # 当前层每一个单词
                for i in range(len(word)):  # 单词的每一位
                    for n in range(26):  # 26个字母
                        new_word = list(word)
                        new_word[i] = chr(ord('a') + n)
                        new_word = ''.join(new_word)
                        if new_word not in wordList: continue
                        if new_word in endVisited: return step + 1
                        if new_word in visited: continue  # 被访问过了
                        next_level.add(new_word)
                        visited.add(new_word)
                        add_step = True
            if add_step: step += 1
            beginVisited = next_level
        return 0

时间复杂度：其实分析起来和方法1一样，但是实际要快一些。O(wordLen×26)。

空间复杂度：和单词列表的长度以及单词的长度有关。

思路是每一次到达一个房间就把该房间里的钥匙收集起来，然后逐一使用已有的钥匙去打开其他的房间，最终看打开的房间的数量是否有n-1个（0号房间默认打开）。

class Solution:
    def canVisitAllRooms(self, rooms) -> bool:
        n = len(rooms)
        reachable = [False] * n
        reachable[0] = True
        keys = set()
        keys = keys.union(rooms[0])

        while keys:
            k = keys.pop()
            if reachable[k]:
                continue
            keys = keys.union(rooms[k])
            reachable[k] = True
        return sum(reachable[1:]) == n - 1

    di = [-1, 1, 0, 0]
    dj = [0, 0, -1, 1]
    def dfs(self, grid, visited, i, j):
        m = len(grid)
        n = len(grid[0])
        if visited[i][j]: return 0
        if grid[i][j] == 0: return 0

        visited[i][j] = True
        cnt = 1
        for k in range(4):
            a = i + self.di[k]
            b = j + self.dj[k]
            if a < 0 or b < 0 or a >= m or b >= n: 
                continue
            if grid[a][b] == 1:
                self.neighbor += 1
            cnt += self.dfs(grid, visited, a, b)

        n = len(grid[0])

        ans = 0
        self.neighbor = 0

                if grid[i][j] == 1 and not visited[i][j]:
                    ans += self.dfs(grid, visited, i, j)

拓扑排序

【labuladong】拓扑排序算法及应用_哔哩哔哩_bilibiliopen in new window

拓扑排序的前提：输入图要保证不能存在环，所以还要有图的环检测算法。

把一个 有向无环图 转成 线性的排序 就叫 拓扑排序。

DFS环检测

DFS是一个回溯的过程，所以无法仅仅使用一个visited数组来同时解决 是否是环 和 是否访问过。这里的做法是使用visited来存储是否访问过，使用onpath来存储是否在路径中，后者的作用就是用来判断环的存在与否。

visited = [False] * n
onPath = [False] * n
hasCycle = False
def dfs(graph, s):
    if onPath[s]: hasCycle = True
    # 访问过的节点就不能再次访问了，有环也要结束遍历了
    if visited[s] or hasCycle: return
    # 将节点s标记为 已经遍历
    visited[s] = True
    # s在路径上
    onPath[s] = True
    for i in graph[s]:
        dfs(graph, i)
    # s访问完了将回溯一下，s就不在路径上了
    onPath[s] = False

BFS环检测

DFS 算法利用 onPath 数组判断是否存在环，BFS 算法借助 indegree 数组记录每个节点的「入度」也可以实现判断是否存在环。

具体做法是：先构图，将图中的入度=0（起点）的点入队，不断弹出队列元素，相当于遍历，然后将其后继结点的入度-1，若入度=0，入队，直到对列为空。遍历队列元素的同时记录已经遍历的节点个数，若个数=总结点个数，则无环，否则有环（环导致环内节点的入度和出度无法变化。）

inDegree = [0] * n
hasCycle = False

def bfs(graph, n):
    q = deque()
    # 将入度=0的节点作为起点，加入队列
    for i in range(n):
        if inDegree[i] == 0:
            q.append(i)
    count = 0
    while q:
        node = q.popleft()
        for i in graph[node]:
            inDegree[i] -= 1
            if inDegree[i] == 0:
                q.append(i)
    if count != n:
        hasCycle = True

class Solution:
    def buildGraph(self, numCourses, prerequisites):
        graph = {}
        for i in range(numCourses):
            graph[i] = []
        for r in prerequisites:
            graph[r[0]].append(r[1])
        return graph

    def checkCycle(self, graph, s):
        if self.onPath[s]: 
            self.hasCycle = True  # 有环
            return
        if self.visited[s]: # 已经被访问过
            return 
        
        self.visited[s] = True
        self.onPath[s] = True
        for i in graph[s]:
            self.checkCycle(graph, i)
        self.onPath[s] = False

        self.visited = [False] * numCourses
        self.onPath = [False] * numCourses

        graph = self.buildGraph(numCourses, prerequisites)
        for s in graph.keys():
            self.checkCycle(graph, s)
        return not self.hasCycle

            count += 1
            for i in graph[node]:
                self.inDegree[i] -= 1
                if self.inDegree[i] == 0:
                    q.append(i)
        if count < numCourses: 
            self.hasCycle = True

    def canFinish(self, numCourses: int, prerequisites: List[List[int]]) -> bool:
        self.hasCycle = False
        self.inDegree = [0] * numCourses

        # 先构建图
        graph = self.buildGraph(numCourses, prerequisites)
        self.ans = []
        self.checkCycle(graph, numCourses, prerequisites)
        return not self.hasCycle

题目：210. 课程表 IIopen in new window

难度：中等

方法1：dfs

用self.ans记录访问的节点，注意，下面代码的构图过程和拓扑排序是相反的，所以为了得到正确的顺序，应该是后序遍历，这里的后序遍历指的是，对于节点v，先遍历其后继节点，最后遍历其本身，所以checkCycle函数的self.ans放置的位置很重要。

class Solution:
    def buildGraph(self, numCourses, prerequisites):
        graph = {}
        for i in range(numCourses):
            graph[i] = []
        
        for r in prerequisites:
            graph[r[0]].append(r[1])
        return graph
    
    def checkCycle(self, graph, s):
        if self.onPath[s]: 
            self.hasCycle = True
            return
        if self.visited[s]: 
            return
        
        self.visited[s] = True
        self.onPath[s] = True
        # label1
        
        for i in graph[s]:
            self.checkCycle(graph, i)
        self.onPath[s] = False
        self.ans.append(s)  # 只能放在这里，不能放在 # label1 处
    
    def findOrder(self, numCourses: int, prerequisites: List[List[int]]) -> List[int]:
        self.hasCycle = False
        self.visited = [False] * numCourses
        self.onPath = [False] * numCourses

        graph = self.buildGraph(numCourses, prerequisites)
        
        self.ans = []
        for i in range(numCourses):
            self.checkCycle(graph, i)
        if self.hasCycle:
            return []
        return self.ans

方法2：bfs

由于建图的过程中，将应该安排靠后的课程作为了有向边的起点，安排靠前的课程作为了有向边的终点，所以这样得出来的是课程的安排的逆序，最终要逆序一下。

class Solution:
    def buildGraph(self, numCourses, prerequisites):
        graph = {}
        for i in range(numCourses):
            graph[i] = []
        for r in prerequisites:
            graph[r[0]].append(r[1])
        return graph

    def checkCycle(self, graph, numCourses, prerequisites):
        # 计算每个节点的入度
        for r in prerequisites:
            from_node = r[0]
            to_node = r[1]
            self.inDegree[to_node] += 1
        # 将入度=0的节点作为入口
        q = deque()
        for i in range(numCourses):
            if self.inDegree[i] == 0:
                q.append(i)
        
        count = 0
        while q:
            node = q.popleft()
            self.ans.append(node)
            count += 1
            for i in graph[node]:
                self.inDegree[i] -= 1
                if self.inDegree[i] == 0:
                    q.append(i)
        if count < numCourses: 
            self.hasCycle = True

    def findOrder(self, numCourses: int, prerequisites: List[List[int]]) -> bool:
        self.hasCycle = False
        self.inDegree = [0] * numCourses

        # 先构建图
        graph = self.buildGraph(numCourses, prerequisites)
        self.ans = []
        self.checkCycle(graph, numCourses, prerequisites)
        if self.hasCycle: return []
        return self.ans[::-1]
